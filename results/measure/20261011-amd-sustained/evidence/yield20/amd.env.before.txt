# amd — bare-metal 4-node efs cluster on amd_efs (amd-395; Ubuntu 26.04,
# x86_64, one NVMe filesystem; /data1/efs is the cluster root.
# Same shape as the nuc cluster: four efsd processes on
# 127.0.0.1 told apart by port. NOTE: hermes/lmstudio also live on this box;
# the quota below is logical (usage grows only with actual test data) — lower
# it if the box needs headroom.
HOST=amd_efs
BASE=/data1/efs               # src, logs, mounts, devops live here
PORTS=(17432 17433 17434 17435)  # node ids 1..4; seed = ${PORTS[0]}
STORDIRS=(/data1/efs/n1 /data1/efs/n2 /data1/efs/n3 /data1/efs/n4)
QUOTA=37.5G                     # per node; 4x37.5G raw, 100G usable (2+1)

# AMD acceptance: strict Soft-RoCE and direct storage I/O.
EFS_TRANSPORT=${EFS_TRANSPORT:-rdma}
EFS_RDMA_DEV=${EFS_RDMA_DEV:-rxe0}
EFS_RDMA_GID_INDEX=${EFS_RDMA_GID_INDEX:-1}
EFS_DIRECT_IO=${EFS_DIRECT_IO:-1}
